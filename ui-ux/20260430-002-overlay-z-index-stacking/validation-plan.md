### 3.1 E2E Test Code Template

```typescript
// e2e/dialog-visibility.spec.ts
import { test, expect } from '@playwright/test'

test.describe('Dialog Z-Index 层级验证', () => {
  test('Dialog 打开后内容不被蒙层遮挡', async ({ page }) => {
    // 1. 导航到用户管理页面
    await page.goto('/users')

    // 2. 点击新增按钮打开 Dialog
    await page.click('button:has-text("新增")')

    // 3. 验证 Dialog 内容可见
    const dialog = page.locator('.fixed.z-modal')
    await expect(dialog).toBeVisible()

    // 4. 验证 Dialog 标题可读取（内容未被遮挡）
    await expect(page.locator('h2')).toContainText(/新增|编辑/)

    // 5. 验证 Dialog 内按钮可点击
    const submitBtn = page.locator('button:has-text("确定")')
    await expect(submitBtn).toBeVisible()
    await expect(submitBtn).toBeEnabled()
  })
})
```
