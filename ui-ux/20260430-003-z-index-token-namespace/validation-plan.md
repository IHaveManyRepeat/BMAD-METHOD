## 4. Static Analysis

### 4.1 构建产物断言脚本

```bash
#!/bin/bash
# admin-frontend/scripts/verify-css-tokens.sh
# 用途：验证 @theme 定义的 token 在编译产物中生成了对应工具类

set -euo pipefail

DIST_DIR="admin-frontend/dist"
if [ ! -d "$DIST_DIR" ]; then
  echo "ERROR: dist/ not found. Run 'pnpm build' first."
  exit 1
fi

CSS_FILE=$(find "$DIST_DIR/assets" -name "index-*.css" | head -1)
if [ -z "$CSS_FILE" ]; then
  echo "ERROR: No compiled CSS found in dist/assets/"
  exit 1
fi

echo "Checking CSS tokens in: $CSS_FILE"
echo ""

# z-index tokens
for token in base dropdown overlay modal toast; do
  if grep -q "\.z-${token}" "$CSS_FILE"; then
    VALUE=$(grep -oP "\.z-${token}\{z-index:\K[^}]+" "$CSS_FILE")
    echo "  PASS: .z-${token} → z-index: ${VALUE}"
  else
    echo "  FAIL: .z-${token} not found in compiled CSS"
    echo "        Check @theme --z-index-${token} in src/index.css"
    exit 1
  fi
done

echo ""
echo "All CSS tokens verified successfully."
```
